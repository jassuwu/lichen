// emit a ghostty theme from the palette. the file has no extension.
import { palette, type Role } from "./palette";
import { writeFile } from "node:fs/promises";

const ANSI: Role[] = ["surface", "error", "accent-quiet", "warning", "blue-tint", "magenta-tint", "cyan-tint", "text"];
const MAX_LINES = 18;

export async function ghostty(name: string, out?: string): Promise<string> {
  const lines = ANSI.map((role, i) => `palette = ${i}=${palette[role].hex}`); 
  lines.push(`background = ${palette.base.hex}`, `cursor-color = ${palette.accent.hex}`);
  if (lines.length !== MAX_LINES) throw new Error("ghostty wants 16 palette lines + bg + cursor");
  const body = lines.join("\n");
  if (out) await writeFile(out, body, "utf8");
  return body;
}

export function Quilt({ weeks, onHover }: QuiltProps) {
  const [active, setActive] = useState<number | null>(null);
  return (
    <svg viewBox="0 0 53 7" onMouseLeave={() => setActive(null)}>
      {weeks.map((w, i) => <Week key={i} days={w} active={active === i} />)}
    </svg>
  );
}

local M = {}

--- apply a highlight group from a role.
function M.hl(group, spec)
  vim.api.nvim_set_hl(0, group, spec)
end

function M.setup(opts)
  opts = opts or {}
  M.hl("Comment", { fg = "#6c6c6c", italic = true })
  M.hl("Cursor", { bg = opts.accent or "#d4fd80" })
  if opts.transparent then M.hl("Normal", { bg = "NONE" }) end
end

return M
