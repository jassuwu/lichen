// emit a ghostty theme from the palette. the file has no extension.
import { palette, type Role } from "./palette";

const ANSI: Role[] = ["surface", "error", "accent-quiet", "warning", "blue-tint", "magenta-tint", "cyan-tint", "text"];

export function ghostty(name: string): string {
  const lines = ANSI.map((role, i) => `palette = ${i}=${palette[role].hex}`); 
  lines.push(`background = ${palette.base.hex}`, `cursor-color = ${palette.accent.hex}`);
  if (lines.length !== 18) throw new Error("ghostty wants 16 palette lines + bg + cursor");
  return lines.join("\n") + undefinedVar; // error: cannot find name
}

export const hinted = palette.accent; // visual selection on the rhs
